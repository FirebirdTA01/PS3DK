cmake_minimum_required(VERSION 3.20)
project(heterogeneous_workload C ASM)
include(ps3-self)
include("${CMAKE_CURRENT_LIST_DIR}/../prof-ext-abi.cmake")
option(SAMPLE_PROFILING "Capture named PPU zones using the candidate SDK profiler" ON)
option(SAMPLE_LIVE "Serve the capture to the PC viewer while it runs (the live link; needs SAMPLE_PROFILING)" OFF)
if(SAMPLE_LIVE AND NOT SAMPLE_PROFILING)
  message(FATAL_ERROR "SAMPLE_LIVE needs SAMPLE_PROFILING: the live link serves the profiler's capture")
endif()
# SPU-lane and RSX-domain zones (design docs r1-r3): ILP32 or LP64; links the UNINSTALLED ext archive of the same data
# model (`make -C sdk/libps3dkprof ext-ilp32` or `ext-lp64`) instead of ps3dkprof, and with SAMPLE_LIVE the live
# members compiled with the ext flags (`make ... ext-live-ilp32` or `ext-live-lp64`, SAMPLE_EXT_LIVE_ARCHIVE).  Every
# member of each archive must carry SAMPLE_ABI's data-model tag (checked at configure time).  OFF: today's build.
option(SAMPLE_PROF_EXT "Record SPU-lane and RSX-domain zones (needs SAMPLE_PROFILING; ILP32 or LP64; the uninstalled ext archive of the same data model)" OFF)
set(SAMPLE_EXT_ARCHIVE "" CACHE FILEPATH "the uninstalled libps3dkprof_ext.a (make ext-ilp32 / ext-lp64); SAMPLE_PROF_EXT only")
set(SAMPLE_EXT_LIVE_ARCHIVE "" CACHE FILEPATH "libps3dkprof_ext_live.a (make ext-live-ilp32 / ext-live-lp64); SAMPLE_PROF_EXT + SAMPLE_LIVE only")
set(SAMPLE_PROF_SRC "${CMAKE_CURRENT_LIST_DIR}/../../../sdk/libps3dkprof/src" CACHE PATH "libps3dkprof sources (internal headers)")
set(SAMPLE_PHASE_SECONDS 8 CACHE STRING "Seconds per workload phase")
set(SAMPLE_SESSION_SECONDS 120 CACHE STRING "Session seconds; 0 explicitly loops until exit")
set(SAMPLE_IDLE_SECONDS 5 CACHE STRING "No-input idle baseline seconds")
foreach(v SAMPLE_PHASE_SECONDS SAMPLE_SESSION_SECONDS SAMPLE_IDLE_SECONDS)
  if(NOT "${${v}}" MATCHES "^[0-9]+$" OR ${v} GREATER 3600)
    message(FATAL_ERROR "${v} must be an integer from 0 to 3600")
  endif()
endforeach()
if(SAMPLE_PHASE_SECONDS EQUAL 0)
  message(FATAL_ERROR "Phase duration must be positive")
endif()
set(SAMPLE_ABI ilp32 CACHE STRING "ilp32 or lp64, matching the selected toolchain")
if(NOT SAMPLE_ABI MATCHES "^(ilp32|lp64)$")
  message(FATAL_ERROR "SAMPLE_ABI must be ilp32 or lp64")
endif()
# SAMPLE_ABI must be the compiler's data model (the toolchain file selects it); main.c asserts it again at compile time
prof_ext_compiled_abi(sample_compiled_abi)
if(NOT SAMPLE_ABI STREQUAL sample_compiled_abi)
  message(FATAL_ERROR "SAMPLE_ABI is ${SAMPLE_ABI} but the compiler's data model is ${sample_compiled_abi} "
    "(select ps3-ppu-toolchain-lp64.cmake and SAMPLE_ABI=lp64 together)")
endif()
if(SAMPLE_ABI STREQUAL "lp64")
  set(sample_abi_lp64 1)
else()
  set(sample_abi_lp64 0)
endif()
set(SAMPLE_BUILD_ID candidate CACHE STRING "Exact build/source identity for captures")
if(SAMPLE_PROF_EXT)
  if(NOT SAMPLE_PROFILING)
    message(FATAL_ERROR "SAMPLE_PROF_EXT needs SAMPLE_PROFILING")
  endif()
  if(NOT EXISTS "${SAMPLE_EXT_ARCHIVE}")
    message(FATAL_ERROR "SAMPLE_PROF_EXT needs SAMPLE_EXT_ARCHIVE (got '${SAMPLE_EXT_ARCHIVE}')")
  endif()
  if(SAMPLE_LIVE AND NOT EXISTS "${SAMPLE_EXT_LIVE_ARCHIVE}")
    message(FATAL_ERROR "SAMPLE_PROF_EXT with SAMPLE_LIVE needs SAMPLE_EXT_LIVE_ARCHIVE (got '${SAMPLE_EXT_LIVE_ARCHIVE}')")
  endif()
  prof_ext_require_abi(${SAMPLE_ABI} "SAMPLE_EXT_ARCHIVE" "${SAMPLE_EXT_ARCHIVE}")
  if(SAMPLE_LIVE)
    prof_ext_require_abi(${SAMPLE_ABI} "SAMPLE_EXT_LIVE_ARCHIVE" "${SAMPLE_EXT_LIVE_ARCHIVE}")
  endif()
endif()
add_executable(heterogeneous-workload source/main.c source/spu_pool.c source/scene.c)
if(SAMPLE_LIVE)
  target_sources(heterogeneous-workload PRIVATE source/live.c)
endif()
if(SAMPLE_PROF_EXT)
  target_sources(heterogeneous-workload PRIVATE source/prof_ext.c source/prof_ext_bind.c source/pool_proto.c)
  target_include_directories(heterogeneous-workload BEFORE PRIVATE "${SAMPLE_PROF_SRC}/../../../sdk/include"
    "${SAMPLE_PROF_SRC}/spu" "${SAMPLE_PROF_SRC}/rsx")
  target_compile_definitions(heterogeneous-workload PRIVATE SAMPLE_PROF_EXT=1)
endif()
target_compile_features(heterogeneous-workload PRIVATE c_std_11)
target_compile_options(heterogeneous-workload PRIVATE -O2 -g -Wall -Wextra -Werror)
# Zone descriptors record __FILE__: store repository-relative paths in
# captures, not the absolute path of the machine that built the sample.
get_filename_component(sample_repo_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
target_compile_options(heterogeneous-workload PRIVATE "-ffile-prefix-map=${sample_repo_root}/=")
target_include_directories(heterogeneous-workload PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../gcm/common" "${PS3DK}/ppu/include")
target_compile_definitions(heterogeneous-workload PRIVATE PS3DK_PROF=$<BOOL:${SAMPLE_PROFILING}>
  SAMPLE_LIVE=$<BOOL:${SAMPLE_LIVE}>
  SAMPLE_PHASE_SECONDS=${SAMPLE_PHASE_SECONDS} SAMPLE_SESSION_SECONDS=${SAMPLE_SESSION_SECONDS}
  SAMPLE_IDLE_SECONDS=${SAMPLE_IDLE_SECONDS} SAMPLE_ABI_LP64=${sample_abi_lp64} SAMPLE_BUILD_ID="${SAMPLE_BUILD_ID}")
if(SAMPLE_ABI STREQUAL "lp64")
  set(sample_lib "${PS3DK}/ppu/lib/lp64")
else()
  set(sample_lib "${PS3DK}/ppu/lib")
endif()
# A candidate profiler outside the SDK (for example a fresh `make -C sdk/libps3dkprof` build): an overlay root with
# ppu/include/ps3dk/prof.h and ppu/lib[/lp64]/libps3dkprof.a, searched before the SDK's own.
set(SAMPLE_PROF_SDK "" CACHE PATH "Overlay root of a candidate profiler (header + archives); empty: the SDK's")
if(SAMPLE_PROF_SDK)
  if(SAMPLE_ABI STREQUAL "lp64")
    set(sample_prof_lib "${SAMPLE_PROF_SDK}/ppu/lib/lp64")
  else()
    set(sample_prof_lib "${SAMPLE_PROF_SDK}/ppu/lib")
  endif()
  if(NOT EXISTS "${SAMPLE_PROF_SDK}/ppu/include/ps3dk/prof.h")
    message(FATAL_ERROR "SAMPLE_PROF_SDK has no ppu/include/ps3dk/prof.h")
  endif()
  target_include_directories(heterogeneous-workload BEFORE PRIVATE "${SAMPLE_PROF_SDK}/ppu/include")
else()
  set(sample_prof_lib "${sample_lib}")
endif()
target_link_options(heterogeneous-workload PRIVATE "-L${sample_prof_lib}" "-L${sample_lib}"
  "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/sample.map")
if(SAMPLE_PROF_EXT)
  if(SAMPLE_LIVE)
    target_link_libraries(heterogeneous-workload PRIVATE "${SAMPLE_EXT_LIVE_ARCHIVE}")
  endif()
  target_link_libraries(heterogeneous-workload PRIVATE "${SAMPLE_EXT_ARCHIVE}")
elseif(SAMPLE_PROFILING)
  if(NOT EXISTS "${sample_prof_lib}/libps3dkprof.a")
    message(FATAL_ERROR "This sample requires a candidate SDK containing libps3dkprof.a; see README")
  endif()
  target_link_libraries(heterogeneous-workload PRIVATE ps3dkprof)
endif()
if(SAMPLE_LIVE)
  target_link_libraries(heterogeneous-workload PRIVATE net netctl_stub)
endif()
target_link_libraries(heterogeneous-workload PRIVATE dbgfont gcm_cmd gcm_sys io sysutil sysmodule rt lv2 m)
set(SAMPLE_SPU_SDK "${PS3DK}" CACHE PATH "SPU SDK root; override only for a partial candidate PPU install")
if(SAMPLE_PROF_EXT)
  # The SPU worker with the lane writer (as ext-probe compiles it).
  ps3_add_spu_image(heterogeneous-workload NAME workload_spu
    SOURCES spu/main.c source/prof_ext.c source/worker_proto.c "${SAMPLE_PROF_SRC}/spu/psp_spu.c" "${SAMPLE_PROF_SRC}/spu/prof_spu_writer.c"
    LIBS sputhread
    CFLAGS "-DPS3DK_PROF=1" "-DSAMPLE_PROF_EXT=1" "-mno-error-reloc" "-I${SAMPLE_PROF_SRC}/spu" "-I${SAMPLE_SPU_SDK}/spu/include"
    LDFLAGS "-L${SAMPLE_SPU_SDK}/spu/lib")
else()
  ps3_add_spu_image(heterogeneous-workload NAME workload_spu SOURCES spu/main.c LIBS sputhread
    CFLAGS "-I${SAMPLE_SPU_SDK}/spu/include" LDFLAGS "-L${SAMPLE_SPU_SDK}/spu/lib")
endif()
ps3_add_cg_shader_rsxcgc(heterogeneous-workload shaders/scene.vcg)
ps3_add_cg_shader_rsxcgc(heterogeneous-workload shaders/light.fcg)
ps3_add_cg_shader_rsxcgc(heterogeneous-workload shaders/heavy.fcg)
ps3_add_self(heterogeneous-workload)
