cmake_minimum_required(VERSION 3.20)
project(heterogeneous_workload C ASM)
include(ps3-self)
option(SAMPLE_PROFILING "Capture named PPU zones using the candidate SDK profiler" ON)
set(SAMPLE_PHASE_SECONDS 8 CACHE STRING "Seconds per workload phase")
set(SAMPLE_SESSION_SECONDS 120 CACHE STRING "Session seconds; 0 explicitly loops until exit")
set(SAMPLE_IDLE_SECONDS 5 CACHE STRING "No-input idle baseline seconds")
foreach(v SAMPLE_PHASE_SECONDS SAMPLE_SESSION_SECONDS SAMPLE_IDLE_SECONDS)
  if(NOT "${${v}}" MATCHES "^[0-9]+$" OR ${v} GREATER 3600)
    message(FATAL_ERROR "${v} must be an integer from 0 to 3600")
  endif()
endforeach()
if(SAMPLE_PHASE_SECONDS EQUAL 0)
  message(FATAL_ERROR "Phase duration must be positive")
endif()
set(SAMPLE_ABI ilp32 CACHE STRING "ilp32 or lp64, matching the selected toolchain")
if(NOT SAMPLE_ABI MATCHES "^(ilp32|lp64)$")
  message(FATAL_ERROR "SAMPLE_ABI must be ilp32 or lp64")
endif()
set(SAMPLE_BUILD_ID candidate CACHE STRING "Exact build/source identity for captures")
add_executable(heterogeneous-workload source/main.c source/spu_pool.c source/scene.c)
target_compile_features(heterogeneous-workload PRIVATE c_std_11)
target_compile_options(heterogeneous-workload PRIVATE -O2 -g -Wall -Wextra -Werror)
# Zone descriptors record __FILE__: store repository-relative paths in
# captures, not the absolute path of the machine that built the sample.
get_filename_component(sample_repo_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
target_compile_options(heterogeneous-workload PRIVATE "-ffile-prefix-map=${sample_repo_root}/=")
target_include_directories(heterogeneous-workload PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../gcm/common" "${PS3DK}/ppu/include")
target_compile_definitions(heterogeneous-workload PRIVATE PS3DK_PROF=$<BOOL:${SAMPLE_PROFILING}>
  SAMPLE_PHASE_SECONDS=${SAMPLE_PHASE_SECONDS} SAMPLE_SESSION_SECONDS=${SAMPLE_SESSION_SECONDS}
  SAMPLE_IDLE_SECONDS=${SAMPLE_IDLE_SECONDS} SAMPLE_ABI="${SAMPLE_ABI}" SAMPLE_BUILD_ID="${SAMPLE_BUILD_ID}")
if(SAMPLE_ABI STREQUAL "lp64")
  set(sample_lib "${PS3DK}/ppu/lib/lp64")
else()
  set(sample_lib "${PS3DK}/ppu/lib")
endif()
target_link_options(heterogeneous-workload PRIVATE "-L${sample_lib}" "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/sample.map")
if(SAMPLE_PROFILING)
  if(NOT EXISTS "${sample_lib}/libps3dkprof.a")
    message(FATAL_ERROR "This sample requires a candidate SDK containing libps3dkprof.a; see README")
  endif()
  target_link_libraries(heterogeneous-workload PRIVATE ps3dkprof)
endif()
target_link_libraries(heterogeneous-workload PRIVATE dbgfont gcm_cmd gcm_sys io sysutil sysmodule rt lv2 m)
set(SAMPLE_SPU_SDK "${PS3DK}" CACHE PATH "SPU SDK root; override only for a partial candidate PPU install")
ps3_add_spu_image(heterogeneous-workload NAME workload_spu SOURCES spu/main.c LIBS sputhread
  CFLAGS "-I${SAMPLE_SPU_SDK}/spu/include" LDFLAGS "-L${SAMPLE_SPU_SDK}/spu/lib")
ps3_add_cg_shader_rsxcgc(heterogeneous-workload shaders/scene.vcg)
ps3_add_cg_shader_rsxcgc(heterogeneous-workload shaders/light.fcg)
ps3_add_cg_shader_rsxcgc(heterogeneous-workload shaders/heavy.fcg)
ps3_add_self(heterogeneous-workload)
