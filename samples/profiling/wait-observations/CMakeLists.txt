cmake_minimum_required(VERSION 3.21)
project(wait_observations C)
include(ps3-self)
if(NOT PS3_PROF_WAITS STREQUAL "OFF" AND NOT PS3_PROF_WAITS STREQUAL "MANUAL")
    message(FATAL_ERROR "This comparison requires PS3_PROF_WAITS=OFF or MANUAL")
endif()
set(WAIT_BUILD_ID "" CACHE STRING "Pinned build identity (required)")
if(NOT WAIT_BUILD_ID MATCHES "^[0-9a-f]+$" OR WAIT_BUILD_ID STREQUAL "")
    message(FATAL_ERROR "WAIT_BUILD_ID must be a nonempty hexadecimal identity")
endif()
add_executable(wait-observations source/main.c)
target_compile_features(wait-observations PRIVATE c_std_11)
target_compile_options(wait-observations PRIVATE -O2 -g -Wall -Wextra -Werror)
target_compile_definitions(wait-observations PRIVATE PS3DK_PROF=1 "WAIT_BUILD_ID=\"${WAIT_BUILD_ID}\"")
if(PS3_PROF_WAITS STREQUAL "MANUAL")
    target_compile_definitions(wait-observations PRIVATE WAIT_OBSERVED=1)
else()
    target_compile_definitions(wait-observations PRIVATE WAIT_OBSERVED=0)
    target_link_libraries(wait-observations PRIVATE ps3dkprof)
endif()
target_link_libraries(wait-observations PRIVATE rt lv2 m)
ps3_add_self(wait-observations)
