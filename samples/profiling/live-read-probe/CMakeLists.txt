cmake_minimum_required(VERSION 3.20)
project(live_read_probe C)
include(ps3-self)
# Live link hardware gate 0 (docs/design/ps3dk-prof-viewer.md, Phase 2):
# reads the profiler's capture file while the file sink still has it open.
# Needs an SDK containing libps3dkprof.a.
set(SAMPLE_ABI ilp32 CACHE STRING "ilp32 or lp64, matching the selected toolchain")
if(NOT SAMPLE_ABI MATCHES "^(ilp32|lp64)$")
  message(FATAL_ERROR "SAMPLE_ABI must be ilp32 or lp64")
endif()
add_executable(live-read-probe source/main.c)
target_compile_features(live-read-probe PRIVATE c_std_11)
target_compile_options(live-read-probe PRIVATE -O2 -g -Wall -Wextra -Werror)
option(PROBE_CONTROL "Build the matched control: same workload, no reader thread" OFF)
target_compile_definitions(live-read-probe PRIVATE PS3DK_PROF=1 PROBE_CONTROL=$<BOOL:${PROBE_CONTROL}>)
target_include_directories(live-read-probe PRIVATE "${PS3DK}/ppu/include")
if(SAMPLE_ABI STREQUAL "lp64")
  set(probe_lib "${PS3DK}/ppu/lib/lp64")
else()
  set(probe_lib "${PS3DK}/ppu/lib")
endif()
if(NOT EXISTS "${probe_lib}/libps3dkprof.a")
  message(FATAL_ERROR "This probe requires an SDK containing libps3dkprof.a")
endif()
target_link_options(live-read-probe PRIVATE "-L${probe_lib}")
target_link_libraries(live-read-probe PRIVATE ps3dkprof rt lv2 m)
ps3_add_self(live-read-probe)
