# spurs-status - every spurs-suite row in one program, with a live
# on-screen status board (cellDbgFont).  Meant for checking SPURS on
# hardware: the rows are the regression suite's own sources, linked in
# as entry functions, so each new row shows up here automatically.
#
# Build:
#   cmake -S . -B build -DCMAKE_TOOLCHAIN_FILE=<repo>/cmake/ps3-ppu-toolchain.cmake
#   cmake --build build
cmake_minimum_required(VERSION 3.20)
project(spurs-status C CXX ASM)

include(ps3-self)

get_filename_component(SUITE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/regression/spurs-suite" ABSOLUTE)
set(SUITE_EMBEDDED ON)
include(${SUITE_DIR}/suite.cmake)
foreach(_row IN LISTS SUITE_ROW_DIRS)
    add_subdirectory(${SUITE_DIR}/rows/${_row} rows/${_row})
endforeach()

# the run table: one entry function per row, in run order
get_property(_rows GLOBAL PROPERTY SUITE_ROWS)
get_property(_row_libs GLOBAL PROPERTY SUITE_ROW_LIBS)
set(_decls "")
set(_table "")
foreach(_row IN LISTS _rows)
    string(MAKE_C_IDENTIFIER "${_row}" _id)
    string(APPEND _decls "int suite_entry_${_id}();\n")
    string(APPEND _table "    { \"${_row}\", suite_entry_${_id} },\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/suite_rows.h.new"
    "/* generated by samples/spurs/spurs-status/CMakeLists.txt */\n${_decls}"
    "static const struct { const char *name; int (*entry)(); } kRows[] = {\n${_table}};\n")
configure_file("${CMAKE_CURRENT_BINARY_DIR}/suite_rows.h.new" "${CMAKE_CURRENT_BINARY_DIR}/suite_rows.h" COPYONLY)

add_executable(spurs-status source/main.cpp)
target_compile_features(spurs-status PRIVATE cxx_std_17)
target_compile_options(spurs-status PRIVATE -Wall -Wextra)
target_include_directories(spurs-status PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${SUITE_DIR}/ppu")
foreach(_row IN LISTS _rows)
    target_sources(spurs-status PRIVATE $<TARGET_OBJECTS:${_row}>)
endforeach()
list(REMOVE_DUPLICATES _row_libs)
target_link_libraries(spurs-status PRIVATE
    ${_row_libs} dbgfont gcm_cmd gcm_sys io ${SUITE_PPU_LIBS})

ps3_add_self(spurs-status)
