cmake_minimum_required(VERSION 3.20)
project(hello-ppu-gcm-frame-pacing C)
include(ps3-self)
add_executable(hello-ppu-gcm-frame-pacing source/main.c)
target_compile_features(hello-ppu-gcm-frame-pacing PRIVATE c_std_11)
target_compile_options(hello-ppu-gcm-frame-pacing PRIVATE -O2 -Wall -Wextra -Werror)
set(PS3DK_GCM_ARCHIVE "" CACHE FILEPATH "Optional private libgcm_cmd archive")
if(PS3DK_GCM_ARCHIVE)
    target_link_libraries(hello-ppu-gcm-frame-pacing PRIVATE "${PS3DK_GCM_ARCHIVE}")
else()
    target_link_libraries(hello-ppu-gcm-frame-pacing PRIVATE gcm_cmd)
endif()
target_link_libraries(hello-ppu-gcm-frame-pacing PRIVATE gcm_sys sysutil_stub sysutil rt lv2)
ps3_add_self(hello-ppu-gcm-frame-pacing)
ps3_add_pkg(hello-ppu-gcm-frame-pacing
    APPID "GCMP00001" CONTENTID "UP0001-GCMP00001_00-PS3DKSAMPLES0000")
