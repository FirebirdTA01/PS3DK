cmake_minimum_required(VERSION 3.20)
project(hello-ppu-gcm-report-copy-ordering C ASM)
set(ORDERING_CLAIM "" CACHE STRING "Required single claim: A1, A2 or B2")
set(claims A1 A2 B2)
list(FIND claims "${ORDERING_CLAIM}" claim_id)
if(claim_id LESS 0)
    message(FATAL_ERROR "Set -DORDERING_CLAIM=A1, A2 or B2; one claim per executable")
endif()
set(target hello-ppu-gcm-report-copy-${ORDERING_CLAIM})
include(ps3-self)
add_executable(${target} source/main.c)
target_compile_features(${target} PRIVATE c_std_11)
target_compile_options(${target} PRIVATE -O2 -Wall -Wextra -Werror)
target_compile_definitions(${target} PRIVATE ORDERING_CLAIM=${claim_id})
target_link_libraries(${target} PRIVATE gcm_cmd gcm_sys io sysutil sysutil_stub sysmodule rt lv2 m)
ps3_add_cg_shader_rsxcgc(${target} ../hello-ppu-cellgcm-state-tour/shaders/vpshader.vcg)
ps3_add_cg_shader_rsxcgc(${target} ../hello-ppu-cellgcm-state-tour/shaders/fpshader.fcg)
ps3_add_self(${target})
