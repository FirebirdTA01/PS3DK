cmake_minimum_required(VERSION 3.20)
project(ds3-bench C CXX)
include(ps3-self)
add_executable(ds3-bench source/main.cpp source/ds3_bench.cpp source/pad_lifecycle.cpp source/display.c)
target_compile_features(ds3-bench PRIVATE cxx_std_17 c_std_11)
target_compile_options(ds3-bench PRIVATE -O2 -Wall -Wextra -Werror)
target_include_directories(ds3-bench PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../gcm/common")
target_link_libraries(ds3-bench PRIVATE dbgfont gcm_cmd gcm_sys io sysutil_stub sysutil rt lv2 m)
ps3_add_self(ds3-bench OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
