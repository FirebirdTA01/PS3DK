cmake_minimum_required(VERSION 3.20)
project(hello-ppu-audio-multistream C)
include(ps3-self)
add_executable(hello-ppu-audio-multistream source/main.c)
target_compile_features(hello-ppu-audio-multistream PRIVATE c_std_11)
target_compile_options(hello-ppu-audio-multistream PRIVATE -O2 -Wall -Wextra -Werror)
set(PS3DK_GCM_ARCHIVE "" CACHE FILEPATH "Optional private libgcm_cmd archive")
if(PS3DK_GCM_ARCHIVE)
    target_link_libraries(hello-ppu-audio-multistream PRIVATE "${PS3DK_GCM_ARCHIVE}")
else()
    target_link_libraries(hello-ppu-audio-multistream PRIVATE gcm_cmd)
endif()
target_link_libraries(hello-ppu-audio-multistream PRIVATE audio gcm_sys sysutil_stub sysutil m rt lv2)
ps3_add_self(hello-ppu-audio-multistream)
ps3_add_pkg(hello-ppu-audio-multistream
    APPID "AUDM00001" CONTENTID "UP0001-AUDM00001_00-PS3DKSAMPLES0000")
