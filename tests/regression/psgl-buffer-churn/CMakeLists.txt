cmake_minimum_required(VERSION 3.20)
project(psgl_buffer_churn C)
option(PS3TC_REGRESSION_HOST_STUB "Host cannot execute PSGL buffer churn" OFF)
if(PS3TC_REGRESSION_HOST_STUB)
    return()
endif()
include(ps3-self)
foreach(abi ilp32 lp64)
    set(target psgl-buffer-churn-${abi})
    add_executable(${target} main.c)
    target_compile_features(${target} PRIVATE c_std_11)
    if(abi STREQUAL "lp64")
        target_compile_options(${target} PRIVATE -mlp64)
        target_link_options(${target} PRIVATE -mlp64)
    endif()
    target_link_libraries(${target} PRIVATE PSGL cgb fs_stub gcm_cmd gcm_sys rsx sysutil_stub sysutil m)
    ps3_add_self(${target})
endforeach()
