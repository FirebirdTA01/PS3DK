cmake_minimum_required(VERSION 3.20)
project(entropy_provider C CXX)
option(PS3TC_REGRESSION_HOST_STUB "Host cannot validate the PS3 entropy service" OFF)
if(PS3TC_REGRESSION_HOST_STUB)
    return()
endif()
include(ps3-self)
foreach(abi ilp32 lp64)
    foreach(mode success error arc4fail)
        add_executable(entropy-${mode}-${abi} main.cpp)
        target_compile_features(entropy-${mode}-${abi} PRIVATE cxx_std_17)
        target_compile_options(entropy-${mode}-${abi} PRIVATE -Wall -Wextra)
        if(abi STREQUAL "lp64")
            target_compile_options(entropy-${mode}-${abi} PRIVATE -mlp64)
            target_link_options(entropy-${mode}-${abi} PRIVATE -mlp64)
        endif()
        if(mode STREQUAL "error")
            target_compile_definitions(entropy-${mode}-${abi} PRIVATE TEST_PROVIDER_ERROR=1)
            target_link_options(entropy-${mode}-${abi} PRIVATE -Wl,--wrap=sys_get_random_number)
        elseif(mode STREQUAL "arc4fail")
            target_compile_definitions(entropy-${mode}-${abi} PRIVATE TEST_ARC4_FAILURE=1)
            target_link_options(entropy-${mode}-${abi} PRIVATE -Wl,--wrap=getentropy)
        endif()
        target_link_options(entropy-${mode}-${abi} PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/entropy-${mode}-${abi}.map")
        ps3_add_self(entropy-${mode}-${abi})
    endforeach()
endforeach()
