cmake_minimum_required(VERSION 3.20)
project(ppu_compare_carry C)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Configure this dual-ABI witness with the ILP32 toolchain")
endif()
if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_CURRENT_BINARY_DIR)
    message(FATAL_ERROR "This witness requires a private out-of-source build")
endif()
set(PS3_SELF_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/packaged")
include(ps3-self)
foreach(abi IN ITEMS ilp32 lp64)
    foreach(opt IN ITEMS 1 2 3)
        set(name ppu-compare-carry-${abi}-O${opt})
        add_library(${name}-functions OBJECT functions.c)
        target_compile_options(${name}-functions PRIVATE -O${opt} -fno-lto -Wall -Wextra -Werror)
        add_executable(${name} main.c $<TARGET_OBJECTS:${name}-functions>)
        target_compile_options(${name} PRIVATE -O0 -fno-lto -Wall -Wextra -Werror)
        target_link_options(${name} PRIVATE -fno-lto)
        target_compile_features(${name} PRIVATE c_std_11)
        target_compile_features(${name}-functions PRIVATE c_std_11)
        if(abi STREQUAL "lp64")
            set(abi_bytes 8)
            target_compile_options(${name}-functions PRIVATE -mlp64)
            target_compile_options(${name} PRIVATE -mlp64)
            target_link_options(${name} PRIVATE -mlp64)
        else()
            set(abi_bytes 4)
        endif()
        target_compile_definitions(${name} PRIVATE PPU_COMPARE_CARRY_ABI_BYTES=${abi_bytes})
        target_compile_definitions(${name}-functions PRIVATE PPU_COMPARE_CARRY_ABI_BYTES=${abi_bytes})
        ps3_add_self(${name})
    endforeach()
endforeach()
