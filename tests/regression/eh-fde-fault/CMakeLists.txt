cmake_minimum_required(VERSION 3.20)
project(eh_fde_fault_probe CXX)

# Not a manifest row: two of the four modes are expected to die (see fault.cpp).  Only meaningful on GCC 16
# (gcc-16.x 0074 reserve / allocation contract).
option(PS3TC_REGRESSION_HOST_STUB "Build with host stubs for CI syntax/link coverage" OFF)
if(PS3TC_REGRESSION_HOST_STUB)
    message(STATUS "eh-fde-fault: skipped under PS3TC_REGRESSION_HOST_STUB")
    return()
endif()
if(CMAKE_CXX_COMPILER_VERSION VERSION_LESS 16)
    message(STATUS "eh-fde-fault: skipped on GCC ${CMAKE_CXX_COMPILER_VERSION} (no 0074 registry)")
    return()
endif()

include(ps3-self)

foreach(abi ilp32 lp64)
    foreach(mode exhaust exhaust_ctl oom oom_ctl)
        string(TOUPPER ${mode} _M)
        string(REPLACE "_" "-" _m ${mode})
        set(t eh-fde-fault-${_m}-${abi})
        add_executable(${t} fault.cpp)
        target_compile_features(${t} PRIVATE cxx_std_17)
        target_compile_definitions(${t} PRIVATE EH_FDE_MODE_${_M})
        target_compile_options(${t} PRIVATE -O2 -Wall -Wextra)
        target_link_options(${t} PRIVATE "-Wl,--wrap=malloc")
        if(abi STREQUAL "lp64")
            target_compile_options(${t} PRIVATE -mlp64)
            target_link_options(${t} PRIVATE -mlp64)
        endif()
        ps3_add_self(${t})
    endforeach()
endforeach()
