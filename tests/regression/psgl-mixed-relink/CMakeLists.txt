cmake_minimum_required(VERSION 3.20)
project(psgl_mixed_relink C ASM)
option(PS3TC_REGRESSION_HOST_STUB "Host cannot run PSGL" OFF)
if(PS3TC_REGRESSION_HOST_STUB)
    return()
endif()
include(ps3-self)
find_program(RSXC NAMES rsx-cg-compiler PATHS "$ENV{PS3DK}/bin" "$ENV{PS3DEV}/bin" REQUIRED)
set(blobs vertex.vpo direct.vpo fragment.fpo yellow.fpo cyan.fpo)
foreach(blob IN LISTS blobs)
    string(REGEX REPLACE "\\.(vpo|fpo)$" ".cg" shader "${blob}")
    if(blob MATCHES "[.]vpo$")
        set(profile sce_vp_rsx)
    else()
        set(profile sce_fp_rsx)
    endif()
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${blob}"
        COMMAND "${RSXC}" -p ${profile} --emit-container "${CMAKE_CURRENT_BINARY_DIR}/${blob}" "${CMAKE_CURRENT_SOURCE_DIR}/${shader}"
        DEPENDS "${shader}" "${RSXC}" VERBATIM)
endforeach()
foreach(abi ilp32 lp64)
    set(target psgl-mixed-relink-${abi})
    add_executable(${target} main.c)
    target_compile_features(${target} PRIVATE c_std_11)
    target_compile_options(${target} PRIVATE -O2 -Wall -Wextra -Werror)
    if(abi STREQUAL "lp64")
        target_compile_options(${target} PRIVATE -mlp64)
        target_link_options(${target} PRIVATE -mlp64)
    endif()
    set(PSGL_ARCHIVE_${abi} "" CACHE FILEPATH "Private PSGL archive")
    if(PSGL_ARCHIVE_${abi})
        target_link_libraries(${target} PRIVATE "${PSGL_ARCHIVE_${abi}}")
    else()
        target_link_libraries(${target} PRIVATE PSGL)
    endif()
    set(GCM_ARCHIVE_${abi} "" CACHE FILEPATH "Private GCM archive")
    if(GCM_ARCHIVE_${abi})
        target_link_libraries(${target} PRIVATE cgb fs_stub "${GCM_ARCHIVE_${abi}}")
    else()
        target_link_libraries(${target} PRIVATE cgb fs_stub gcm_cmd)
    endif()
    target_link_libraries(${target} PRIVATE gcm_sys rsx sysutil_stub sysutil m)
    foreach(blob IN LISTS blobs)
        if(abi STREQUAL "ilp32")
            ps3_bin2s(${target} "${CMAKE_CURRENT_BINARY_DIR}/${blob}")
        else()
            target_sources(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/${blob}.s")
            target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        endif()
    endforeach()
    ps3_add_self(${target})
endforeach()
