cmake_minimum_required(VERSION 3.20)
project(psgl_teardown_finish C ASM)
option(PS3TC_REGRESSION_HOST_STUB "Host cannot execute PSGL teardown" OFF)
if(PS3TC_REGRESSION_HOST_STUB)
    return()
endif()
include(ps3-self)
foreach(abi ilp32 lp64)
    set(PSGL_ARCHIVE_${abi} "" CACHE FILEPATH "Private PSGL archive")
    set(GCM_ARCHIVE_${abi} "" CACHE FILEPATH "Private GCM archive")
    foreach(variant teardown-finish teardown-exit-timeout teardown-exit-flip teardown-exit-shutdown)
        set(target psgl-${variant}-${abi})
        add_executable(${target} main.c)
        target_compile_features(${target} PRIVATE c_std_11)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
        if(variant STREQUAL "teardown-exit-timeout")
            target_compile_definitions(${target} PRIVATE TEARDOWN_EXIT_MODE=1)
        elseif(variant STREQUAL "teardown-exit-flip")
            target_compile_definitions(${target} PRIVATE TEARDOWN_EXIT_MODE=2)
        elseif(variant STREQUAL "teardown-exit-shutdown")
            target_compile_definitions(${target} PRIVATE TEARDOWN_EXIT_MODE=3)
        endif()
        if(abi STREQUAL "lp64")
            target_compile_options(${target} PRIVATE -mlp64)
            target_link_options(${target} PRIVATE -mlp64)
        endif()
        if(PSGL_ARCHIVE_${abi})
            target_link_libraries(${target} PRIVATE "${PSGL_ARCHIVE_${abi}}")
        else()
            target_link_libraries(${target} PRIVATE PSGL)
        endif()
        if(GCM_ARCHIVE_${abi})
            target_link_libraries(${target} PRIVATE cgb fs_stub "${GCM_ARCHIVE_${abi}}")
        else()
            target_link_libraries(${target} PRIVATE cgb fs_stub gcm_cmd)
        endif()
        target_link_libraries(${target} PRIVATE gcm_sys rsx sysutil_stub sysutil m)
        ps3_add_self(${target})
    endforeach()
endforeach()
