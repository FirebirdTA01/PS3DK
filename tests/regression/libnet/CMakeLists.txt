cmake_minimum_required(VERSION 3.20)
project(libnet_regression C)
option(PS3TC_REGRESSION_HOST_STUB "Host syntax builds cannot validate PPU networking" OFF)
if(PS3TC_REGRESSION_HOST_STUB)
    return()
endif()
include(ps3-self)
foreach(abi ilp32 lp64)
    add_executable(libnet-${abi} main.c)
    target_compile_features(libnet-${abi} PRIVATE c_std_11)
    target_compile_options(libnet-${abi} PRIVATE -Wall -Wextra)
    target_link_libraries(libnet-${abi} PRIVATE net_stub sysmodule_stub)
    if(abi STREQUAL "lp64")
        target_compile_options(libnet-${abi} PRIVATE -mlp64)
        target_link_options(libnet-${abi} PRIVATE -mlp64)
    endif()
    ps3_add_self(libnet-${abi})
endforeach()
