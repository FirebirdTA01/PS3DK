cmake_minimum_required(VERSION 3.20)
project(posix_time_regression C)
option(PS3TC_REGRESSION_HOST_STUB "Host cannot validate Lv-2 timing" OFF)
if(PS3TC_REGRESSION_HOST_STUB)
    return()
endif()
include(ps3-self)
foreach(abi ilp32 lp64)
    add_executable(posix-time-${abi} main.c)
    target_compile_features(posix-time-${abi} PRIVATE c_std_11)
    target_compile_options(posix-time-${abi} PRIVATE -Wall -Wextra -Werror=implicit-function-declaration)
    if(abi STREQUAL "lp64")
        target_compile_options(posix-time-${abi} PRIVATE -mlp64)
        target_link_options(posix-time-${abi} PRIVATE -mlp64)
    endif()
    ps3_add_self(posix-time-${abi})
endforeach()
