cmake_minimum_required(VERSION 3.20)
project(psgl_client_scratch C ASM)
option(PS3TC_REGRESSION_HOST_STUB "Host cannot execute PSGL client scratch rows" OFF)
if(PS3TC_REGRESSION_HOST_STUB)
    return()
endif()
include(ps3-self)
find_program(RSXC NAMES rsx-cg-compiler PATHS "$ENV{PS3DK}/bin" "$ENV{PS3DEV}/bin" REQUIRED)
foreach(stage vertex fragment)
    if(stage STREQUAL "vertex")
        set(profile sce_vp_rsx)
        set(suffix vpo)
    else()
        set(profile sce_fp_rsx)
        set(suffix fpo)
    endif()
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${stage}.${suffix}"
        COMMAND "${RSXC}" -p ${profile} --emit-container "${CMAKE_CURRENT_BINARY_DIR}/${stage}.${suffix}"
            "${CMAKE_CURRENT_SOURCE_DIR}/${stage}.cg"
        DEPENDS "${stage}.cg" "${RSXC}" VERBATIM)
endforeach()
set(first_target TRUE)
foreach(mode wrap growth wrap-timeout growth-timeout)
    foreach(abi ilp32 lp64)
        set(target psgl-client-scratch-${mode}-${abi})
        add_executable(${target} main.c)
        target_compile_features(${target} PRIVATE c_std_11)
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
        if(mode MATCHES "growth")
            target_compile_definitions(${target} PRIVATE TEST_GROWTH=1)
        endif()
        if(mode MATCHES "timeout")
            target_compile_definitions(${target} PRIVATE TEST_TIMEOUT=1)
        endif()
        if(abi STREQUAL "lp64")
            target_compile_options(${target} PRIVATE -mlp64)
            target_link_options(${target} PRIVATE -mlp64)
        endif()
        set(PSGL_ARCHIVE_${abi} "" CACHE FILEPATH "Private PSGL archive")
        if(PSGL_ARCHIVE_${abi})
            target_link_libraries(${target} PRIVATE "${PSGL_ARCHIVE_${abi}}")
        else()
            target_link_libraries(${target} PRIVATE PSGL)
        endif()
        target_link_libraries(${target} PRIVATE cgb fs_stub gcm_cmd gcm_sys rsx sysutil_stub sysutil m)
        if(first_target)
            ps3_bin2s(${target} "${CMAKE_CURRENT_BINARY_DIR}/vertex.vpo")
            ps3_bin2s(${target} "${CMAKE_CURRENT_BINARY_DIR}/fragment.fpo")
            set(first_target FALSE)
        else()
            target_sources(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/vertex.vpo.s"
                "${CMAKE_CURRENT_BINARY_DIR}/fragment.fpo.s")
            target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        endif()
        ps3_add_self(${target})
    endforeach()
endforeach()
