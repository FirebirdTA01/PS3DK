cmake_minimum_required(VERSION 3.20)
project(heap_small_blocks C)
option(PS3TC_REGRESSION_HOST_STUB "Use the host heap test driver instead" OFF)
if(PS3TC_REGRESSION_HOST_STUB)
    return()
endif()
include(ps3-self)
foreach(abi ilp32 lp64)
    set(target heap-small-blocks-${abi})
    add_executable(${target} ../../host/fixtures/heap-small-block.c)
    target_compile_features(${target} PRIVATE c_std_11)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    if(abi STREQUAL "lp64")
        target_compile_options(${target} PRIVATE -mlp64)
        target_link_options(${target} PRIVATE -mlp64)
    endif()
    ps3_add_self(${target})
endforeach()
