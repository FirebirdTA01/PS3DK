cmake_minimum_required(VERSION 3.20)
project(profiler_thread_exit C CXX)
option(PS3TC_REGRESSION_HOST_STUB "Host cannot validate LV2 thread exit or weak descriptors" OFF)
if(PS3TC_REGRESSION_HOST_STUB)
    return()
endif()
include(ps3-self)
option(PROFILER_EXIT_CANDIDATE_FAULT_TESTS "Build candidate-only, fault-injected runtime tests (not manifest rows)" OFF)
set(_runtime "${CMAKE_CURRENT_SOURCE_DIR}/../../../runtime/lv2/librt")
set(_variants installed)
if(PROFILER_EXIT_CANDIDATE_FAULT_TESTS)
    list(APPEND _variants candidate-fault)
    set_source_files_properties("${_runtime}/pthread.c" PROPERTIES
        COMPILE_OPTIONS "-fexceptions;-include;${CMAKE_CURRENT_SOURCE_DIR}/fault.h")
endif()
foreach(abi ilp32 lp64)
    foreach(hook absent present)
      foreach(variant IN LISTS _variants)
        set(target profiler-thread-exit-${abi}-${hook})
        if(variant STREQUAL "candidate-fault")
            string(APPEND target -candidate-fault)
            add_executable(${target} main.cpp "${_runtime}/pthread.c")
            target_include_directories(${target} PRIVATE "${_runtime}")
            target_compile_definitions(${target} PRIVATE PROFILER_EXIT_FAULT_TEST=1)
        else()
            # Manifest rows exercise the installed librt, without source overrides.
            add_executable(${target} main.cpp)
            target_compile_definitions(${target} PRIVATE PROFILER_EXIT_FAULT_TEST=0)
        endif()
        target_compile_features(${target} PRIVATE c_std_11 cxx_std_17)
        target_compile_options(${target} PRIVATE -O2 -g -Wall -Wextra -fno-instrument-functions)
        if(hook STREQUAL "present")
            target_compile_definitions(${target} PRIVATE EXPECT_EXIT_HOOK=1)
        else()
            target_compile_definitions(${target} PRIVATE EXPECT_EXIT_HOOK=0)
        endif()
        if(abi STREQUAL "lp64")
            target_compile_options(${target} PRIVATE -mlp64)
            target_link_options(${target} PRIVATE -mlp64)
            target_link_options(${target} PRIVATE "-L${PS3DK}/ppu/lib/lp64")
        else()
            target_link_options(${target} PRIVATE "-L${PS3DK}/ppu/lib")
        endif()
        # Keep this explicit: CMake elides implicit link directories, allowing
        # GCC's toolchain mirror to hide a newly installed SDK runtime archive.
        target_link_options(${target} PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/${target}.map,--cref")
        ps3_add_self(${target})
      endforeach()
    endforeach()
endforeach()
