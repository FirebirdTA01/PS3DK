# rsx-cg-compiler — Cg → NV40 compiler
#
# Self-contained build: the Cg front end (from FirebirdTA01's Vita Cg
# compiler, same author, MIT) lives under tools/rsx-cg-compiler/src/cg_frontend/.

cmake_minimum_required(VERSION 3.16)
project(rsx-cg-compiler
    LANGUAGES CXX
    DESCRIPTION "PS3 Cg-to-RSX (NV40) shader compiler (CgBinaryProgram compat)")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

if(NOT MSVC)
    add_compile_options(
        -Wall -Wextra
        -Wno-unused-parameter
        -Wno-unused-variable
        -Wno-unused-but-set-variable
        -Wno-unused-function
        -Wno-missing-field-initializers
        -Wno-sign-compare
        -Wno-reorder
        -Wno-switch)
else()
    # /Brepro: cl and link write a hash of the output where they would write
    # the time (the PE TimeDateStamp and the debug-directory stamp), so two
    # builds of one commit give the same .exe.  Not in Debug, where it would
    # turn off incremental linking.
    add_compile_options($<$<NOT:$<CONFIG:Debug>>:/Brepro>)
    add_link_options($<$<NOT:$<CONFIG:Debug>>:/Brepro>)
endif()

set(SRC_DIR   ${CMAKE_CURRENT_SOURCE_DIR}/src)
set(CG_FRONTEND_DIR ${SRC_DIR}/cg_frontend)

# ----------------------------------------------------------------------------
# Version stamp — sourced from the repo-wide scripts/version.sh so the
# rsx-cg-compiler binary reports the same version as the rest of the SDK.
# ----------------------------------------------------------------------------

set(_repo_root ${CMAKE_CURRENT_SOURCE_DIR}/../..)
set(_version_sh ${_repo_root}/scripts/version.sh)

# Find a bash interpreter explicitly.  On Linux execute_process can launch
# a .sh via its shebang, but on Windows CreateProcess refuses .sh outright
# even inside MSYS2 — we have to invoke bash by name.
find_program(_bash_exe NAMES bash REQUIRED)

# Run the generator in --format=cmake mode and include the result, which
# defines PS3SDK_VERSION* variables.  The file is regenerated on every
# configure step so a new commit picks up immediately.
set(_version_cmake ${CMAKE_CURRENT_BINARY_DIR}/ps3sdk_version.cmake)
execute_process(
    COMMAND ${_bash_exe} ${_version_sh} --format=cmake
    OUTPUT_FILE ${_version_cmake}
    WORKING_DIRECTORY ${_repo_root}
    RESULT_VARIABLE _version_rc)
if(NOT _version_rc EQUAL 0)
    message(FATAL_ERROR "scripts/version.sh failed (rc=${_version_rc})")
endif()
include(${_version_cmake})
message(STATUS "rsx-cg-compiler version: ${PS3SDK_VERSION}")

configure_file(
    ${SRC_DIR}/version.h.in
    ${CMAKE_CURRENT_BINARY_DIR}/version.h
    @ONLY)

# ----------------------------------------------------------------------------
# Embedded builtin Cg header (generated by the front end's embed_builtin_header.py).
# ----------------------------------------------------------------------------

find_package(Python3 REQUIRED COMPONENTS Interpreter)

set(EMBED_INPUT  ${CG_FRONTEND_DIR}/builtin_shader_header.h)
set(EMBED_OUTPUT ${CG_FRONTEND_DIR}/builtin_shader_header_embedded.inc)
set(EMBED_SCRIPT ${CG_FRONTEND_DIR}/embed_builtin_header.py)

add_custom_command(
    OUTPUT  ${EMBED_OUTPUT}
    COMMAND ${Python3_EXECUTABLE} ${EMBED_SCRIPT} ${EMBED_INPUT} ${EMBED_OUTPUT}
    DEPENDS ${EMBED_INPUT} ${EMBED_SCRIPT}
    COMMENT "Embedding builtin shader header (${EMBED_OUTPUT})"
    VERBATIM)

# ----------------------------------------------------------------------------
# Source lists.
# ----------------------------------------------------------------------------

# Cg front-end sources (from the Vita Cg compiler).
set(SRC_CG_FRONTEND
    ${CG_FRONTEND_DIR}/common/errorHandler.cpp
    ${CG_FRONTEND_DIR}/common/memoryPool.cpp
    ${CG_FRONTEND_DIR}/frontend/preprocessor.cpp
    ${CG_FRONTEND_DIR}/frontend/lexer.cpp
    ${CG_FRONTEND_DIR}/frontend/parser.cpp
    ${CG_FRONTEND_DIR}/frontend/semantic.cpp
    ${CG_FRONTEND_DIR}/frontend/symbol_table.cpp
    ${CG_FRONTEND_DIR}/frontend/types.cpp
    ${CG_FRONTEND_DIR}/builtin_shader_header.cpp
)

# Front-end IR layer — hardware-independent IRModule built from AST.
# The Vita compiler's USSE back-end (src/backend/, src/gxp/) is not used:
# NV40 lowering lives in the SRC_NV40 list below.
set(SRC_CG_FRONTEND_IR
    ${CG_FRONTEND_DIR}/ir/ir.cpp
    ${CG_FRONTEND_DIR}/ir/ir_builder.cpp
    ${CG_FRONTEND_DIR}/ir/ir_passes.cpp
)

# NV40 back-end (this tree).  Register-file / opcode headers are adopted
# from Mesa/Nouveau via PSL1GHT cgcomp; per-stage lowering is authored here.
set(NV40_DIR ${SRC_DIR}/nv40)
set(SRC_NV40
    ${NV40_DIR}/nv40_emit.cpp
    ${NV40_DIR}/nv40_general_lowering.cpp
    ${NV40_DIR}/nv40_vp_assembler.cpp
    ${NV40_DIR}/nv40_fp_assembler.cpp
    ${NV40_DIR}/nv40_discard_guards.cpp
    ${NV40_DIR}/nv40_if_convert.cpp
)

# Cg-binary container emitter — wraps NV40 ucode in a CgBinaryProgram
# (.fpo / .vpo) so the runtime can consume it via cellGcmCgInitProgram.
set(SRC_CONTAINER
    ${SRC_DIR}/cg_container_fp.cpp
    ${SRC_DIR}/cg_container_vp.cpp
)

# Driver / entry point, and the named-extension registry it consults.
set(SRC_MAIN
    ${SRC_DIR}/extensions.cpp
    ${SRC_DIR}/main.cpp
)

add_executable(rsx-cg-compiler
    ${SRC_CG_FRONTEND}
    ${SRC_CG_FRONTEND_IR}
    ${SRC_NV40}
    ${SRC_CONTAINER}
    ${SRC_MAIN}
    ${EMBED_OUTPUT}
)

target_include_directories(rsx-cg-compiler PRIVATE
    ${CMAKE_CURRENT_BINARY_DIR}    # generated version.h — wins over cg_frontend/common/version.h
    ${CG_FRONTEND_DIR}
    ${CG_FRONTEND_DIR}/common
    ${CG_FRONTEND_DIR}/frontend
    ${CG_FRONTEND_DIR}/ir
    ${SRC_DIR}
    ${NV40_DIR}
)

# ----------------------------------------------------------------------------
# cgnv2elf — standalone converter that packs .vpo / .fpo containers into an
# ELF shader archive.  Plain C++17; shares only the generated version.h.
# ----------------------------------------------------------------------------

add_executable(cgnv2elf
    ${SRC_DIR}/cgnv2elf/cgnv2elf.cpp
)

add_executable(fp-register-remap-test EXCLUDE_FROM_ALL tests/fp_register_remap_test.cpp)
target_include_directories(fp-register-remap-test PRIVATE ${SRC_DIR})

target_include_directories(cgnv2elf PRIVATE
    ${CMAKE_CURRENT_BINARY_DIR}    # generated version.h
)
