cmake_minimum_required(VERSION 3.20)
project(dxt_encoder_example C)
add_library(dxt_encoder STATIC dxt.c)
target_compile_features(dxt_encoder PUBLIC c_std_11)
target_include_directories(dxt_encoder PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(dxt-example example.c)
target_link_libraries(dxt-example PRIVATE dxt_encoder)
include(CTest)
if(BUILD_TESTING)
    add_executable(dxt-check ../../tests/sdk/dxt-encoder-test.c)
    target_link_libraries(dxt-check PRIVATE dxt_encoder)
    add_test(NAME dxt-format COMMAND dxt-check)
    add_test(NAME dxt-example-check COMMAND dxt-example)
endif()
